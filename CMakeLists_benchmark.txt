cmake_minimum_required(VERSION 3.10)
project(idcu_benchmark C)
set(CMAKE_C_STANDARD 99)

if(MSVC)
    add_compile_options(/W4)
else()
    add_compile_options(-O2 -Wall -Wextra)
endif()

include_directories(${PROJECT_SOURCE_DIR} include kernel modules)

file(GLOB_RECURSE BENCH_SRCS
    benchmark.c
    kernel/coroutine.c
    kernel/msg_bus.c
    kernel/memory_pool.c
    kernel/lock.c
    kernel/log.c
    kernel/atomic.c
    kernel/context.c
)

add_executable(idcu_benchmark ${BENCH_SRCS})
set_target_properties(idcu_benchmark PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/out
)

if(WIN32)
    target_link_libraries(idcu_benchmark PRIVATE)
else()
    target_link_libraries(idcu_benchmark PRIVATE pthread)
endif()
